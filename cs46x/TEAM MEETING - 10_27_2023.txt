import os
import random
from pywavefront import Wavefront
from PIL import Image
# Function to extract 2D images from a 3D object at random orientations
def extract_2d_images(obj_file, output_dir, num_images):
    # Load the OBJ file
    obj = Wavefront(obj_file)
    # Create the output directory if it doesn't exist
    os.makedirs(output_dir, exist_ok=True)
    # Extract random 2D images
    for i in range(num_images):
        # Randomly select an orientation
        rotation = (random.uniform(0, 360), random.uniform(0, 360), random.uniform(0, 360))
        # Render the object with the specified rotation
        image = obj.scene.render(rotation=rotation)
        # Save the image
        image_path = os.path.join(output_dir, f"image_{i+1}.png")
        image.save(image_path)
        print(f"Image {i+1} saved.")
# Usage example
